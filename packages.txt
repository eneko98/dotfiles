# Packages this desktop adds, in the order they were added.
#
# A fresh Arch install only provides the baseline: base, base-devel, linux,
# linux-firmware, amd-ucode, mesa, git, neovim, pipewire, sudo. Everything
# listed below is what sits on top of that, one section per component.
#
# Install everything with:
#   grep -vE '^\s*(#|$)' packages.txt | sudo pacman -S --needed -
#
# The comment/blank-line filtering is needed because pacman treats every line
# it reads from stdin as a package name, comments included.

# 01 - graphics stack and fonts
# vulkan-radeon is *not* required by Hyprland (aquamarine only needs mesa/EGL),
# but it is the recommended driver for AMD on Wayland and any future Vulkan
# application needs it, so it is installed up front.
vulkan-radeon
vulkan-icd-loader
fontconfig
ttf-jetbrains-mono-nerd

# 01 - fallback kernel
# Insurance: systemd-boot gets a second, known-good entry to boot from if a
# graphics or kernel change breaks the main one. This package alone is NOT
# enough -- systemd-boot here reads only Type #1 entries, so
# /boot/loader/entries/arch-linux-lts.conf had to be written by hand too.
linux-lts

# 02 - compositor and terminal
hyprland
kitty

# 02 - cursor theme
# Not cosmetic: the system had no cursor theme whatsoever, so no pointer would
# have been drawn. Replaceable when the theme pass happens.
adwaita-cursors
