# Packages this desktop adds, in the order they were added.
#
# A fresh Arch install only provides the baseline: base, base-devel, linux,
# linux-firmware, amd-ucode, mesa, git, neovim, pipewire, sudo. Everything
# listed below is what sits on top of that, one section per component.
#
# Install everything with:
#   grep -vE '^\s*(#|$)' packages.txt | sudo pacman -S --needed -
#
# The comment/blank-line filtering is needed because pacman treats every line
# it reads from stdin as a package name, comments included.

# 01 - graphics stack and fonts
# vulkan-radeon is *not* required by Hyprland (aquamarine only needs mesa/EGL),
# but it is the recommended driver for AMD on Wayland and any future Vulkan
# application needs it, so it is installed up front.
vulkan-radeon
vulkan-icd-loader
fontconfig
ttf-jetbrains-mono-nerd

# 01 - fallback kernel
# Insurance: systemd-boot gets a second, known-good entry to boot from if a
# graphics or kernel change breaks the main one. This package alone is NOT
# enough -- systemd-boot here reads only Type #1 entries, so
# /boot/loader/entries/arch-linux-lts.conf had to be written by hand too.
linux-lts

# 02 - compositor and terminal
hyprland
kitty

# 02 - cursor theme
# Not cosmetic: the system had no cursor theme whatsoever, so no pointer would
# have been drawn. Replaceable when the theme pass happens.
adwaita-cursors

# 03 - zsh and oh-my-zsh
# oh-my-zsh itself is not a package; it is cloned into XDG_DATA_HOME. See the
# README, which also covers the /etc/zsh/zshenv line that makes ZDOTDIR work.
# Only the archlinux plugin ships with oh-my-zsh; the other three come from the
# repos and are sourced by hand from config/zsh/.zshrc.
zsh
zsh-autosuggestions
zsh-syntax-highlighting
zsh-completions

# 04 - session manager
# uwsm wraps the compositor in systemd user units -- wayland-wm@hyprland.desktop.service
# pulled in by graphical-session.target -- instead of leaving it as a bare process, and
# gives SUPER+M a clean "uwsm stop" path out. Started by config/zsh/.zprofile; the tty1
# autologin drop-in around it is in the README.
uwsm

# 05 - wallpaper
# hyprpaper paints the wall behind the windows. Chosen over swaybg because it has
# IPC -- `hyprctl hyprpaper wallpaper ...` changes it live -- and over swww/mpvpaper
# because those are AUR-only and there is no AUR helper yet.
# It is started by the systemd user unit the package ships, not from hyprland.lua:
# the uwsm wiki recommends the unit, and a unit restarts on crash and stops with the
# session. One-time setup: systemctl --user enable hyprpaper.service
hyprpaper

# 08 - desktop essentials: the application launcher
# fuzzel, on SUPER + Space. Toolkit-free -- no GTK, no X11, unlike wofi (GTK) or
# dmenu (X11, needs XWayland). Its colours are generated from the palette by
# scripts/gen-theme.py, which is the fourth consumer and the reason that script
# exists at all.
fuzzel

# 08 - desktop essentials: screenshots
# grim reads pixels from the compositor and writes a PNG, slurp is the "drag a
# rectangle" part, wl-clipboard puts the result on the clipboard so a shot can be
# pasted straight into a chat. All three in extra: no portal, no daemon, 185 KiB
# together. Driven by config/hypr/scripts/screenshot.
grim
slurp
wl-clipboard

# 09 - file manager
# yazi is the file manager, with image, video, PDF and archive previews inside
# the terminal. fd, ripgrep, fzf and zoxide are the surrounding tools: fast find,
# fast grep, fuzzy selection and learned cd. The yazi theme is generated from the
# palette, so scripts/gen-theme.py gains a fifth consumer. xdg-utils is needed
# for the default "open with" behaviour.
yazi
fd
ripgrep
fzf
zoxide
jq
ffmpegthumbnailer
poppler
imagemagick
7zip
chafa
xdg-utils
